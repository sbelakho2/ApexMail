example.py202 ACCEPTED
import requests

resp = requests.post(
    "https://api.apexmail.ee/v1/messages",
    headers={"X-API-Key": "am_live_xxx"},
    json={
        "from": "hello@company.com",
        "to": ["user@example.com"],
        "subject": "Welcome!",
        "html": "<h1>Hello!</h1>",
    },
)
resp.raise_for_status()  # 202 Accepted on success
message = resp.json()["data"]  # {"id": ..., "status": "queued", ...}

# Track delivery
detail = requests.get(
    f"https://api.apexmail.ee/v1/messages/{message['id']}",
    headers={"X-API-Key": "am_live_xxx"},
)
detail.raise_for_status()
print(detail.json()["data"]["status"])  # 'queued'

compliance.pyAUDIT TRAIL
import requests

# Message-level tags are routing/analytics labels for your own reporting.
# They are NOT consent records or lawful-basis evidence — consent and
# processing-basis records live in the compliance workflow (/compliance/),
# not in message tags.
resp = requests.post(
    "https://api.apexmail.ee/v1/messages",
    headers={"X-API-Key": "am_live_xxx"},
    json={
        "from": "hello@company.com",
        "to": ["user@example.com"],
        "subject": "Your monthly statement",
        "html": "<p>Your statement is ready.</p>",
        "tags": ["purpose=transactional", "monthly-statement"],
    },
)
resp.raise_for_status()  # 202 Accepted on success

# Every message's event history (accepted → delivered → opened)
# feeds the audit trail exposed to compliance workflows
message_id = resp.json()["data"]["id"]

events.pyEVENT HISTORY
import requests

# Every message carries its full event history
events = requests.get(
    "https://api.apexmail.ee/v1/events?message_id=msg_2xk9",
    headers={"X-API-Key": "am_live_xxx"},
).json()

for event in events:
    print(event["event_type"], event["timestamp"])
# message.accepted 2026-09-09T10:41:02Z
# message.delivered 2026-09-09T10:41:04Z
# message.opened 2026-09-09T11:02:47Z
